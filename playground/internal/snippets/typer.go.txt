// This is an example of using sleep to slowly output text
// to look as if someone was manually typing the text very fast.
package main

import (
	"fmt"
	"math/rand"
	"time"
)

func main() {
	for _, c := range ipsum {
		fmt.Print(string([]rune{c}))

		dur := time.Duration(rand.Intn(10)+2) * time.Millisecond
		time.Sleep(dur)
	}
}
